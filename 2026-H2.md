  ```sql
  DECLARE @startTime DATETIME2(0) = '2026-08-10 8:00:00', @endTime DATETIME2(0) = '2026-08-10 9:00:00'

  SELECT  [database_name],
          sql_pool_name,
          distributed_statement_id,
          login_name,
          allocated_cpu_time_ms / 1000.0 AS vcore_seconds
  FROM queryinsights.exec_requests_history
  WHERE start_time < @endTime AND end_time > @startTime
  ```
